cmake_minimum_required(VERSION 3.22.1)
project(typeright_gguf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
# Baseline ARMv8: do not require newer phone CPU instructions.
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_compile_options(-O3)
include(FetchContent)
FetchContent_Declare(llama
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/74d4f5b041ad837153b0e90fc864b8290e01d8d5
    URL_HASH SHA256=8dd9fcc7c17f972673960ffdcc7521f52de5a9ec594aa6acebd9dfd8c17c0a5c
)
FetchContent_MakeAvailable(llama)
add_library(typeright_gguf SHARED gguf_jni.cpp)
target_link_libraries(typeright_gguf PRIVATE llama android log)
target_compile_options(typeright_gguf PRIVATE -O3)
target_link_options(typeright_gguf PRIVATE "-Wl,-z,max-page-size=16384")
